var PAIRS = {
  "Brukinsa patient recruited": "Brukinsa patient target",

  "Sales (value)":              "Target (value)"
};

function money(n){
  if(n === null || n === undefined || isNaN(n)) return "—";
  return (Math.round(n*100)/100).toLocaleString("en-GB");
}
function sumOf(list, measure, monthIdx){
  var t = 0, seen = false;
  list.forEach(function(a){
    var m = a.measures[measure];
    if(!m) return;
    var v = (monthIdx === "ytd") ? m.ytd : m.m[monthIdx];
    if(v !== null && v !== undefined && !isNaN(v)){ t += v; seen = true; }
  });
  return seen ? t : null;
}
function seriesOf(list, measure){
  return MONTHS.map(function(_, i){ return sumOf(list, measure, i); });
}
function measureList(){
  var out = [];
  Object.keys(SALES.measureCats).forEach(function(cat){
    SALES.measureCats[cat].forEach(function(m){ out.push({cat:cat, name:m}); });
  });
  return out;
}
function filteredAccounts(){
  var own = document.getElementById("fowner").value;
  var city = document.getElementById("fcity").value;
  return SALES.accounts.filter(function(a){
    if(own && a.ps !== own) return false;
    if(city && a.city !== city) return false;
    return true;
  });
}

/* ---------- filters ---------- */
function fillSelect(el, values, allLabel){
  el.innerHTML = "";
  if(allLabel){ var o = document.createElement("option"); o.value = ""; o.textContent = allLabel; el.appendChild(o); }
  values.forEach(function(v){
    var o = document.createElement("option");
    o.value = (typeof v === "string") ? v : v.value;
    o.textContent = (typeof v === "string") ? v : v.label;
    el.appendChild(o);
  });
}
function buildSalesFilters(){

  var cities = Array.from(new Set(SALES.accounts.map(function(a){ return a.city; }))).sort();
  fillSelect(document.getElementById("fowner"), owners, "All owners");
  fillSelect(document.getElementById("fcity"), cities, "All cities");

  var sel = document.getElementById("fmeasure");
  sel.innerHTML = "";
  Object.keys(SALES.measureCats).forEach(function(cat){
    var g = document.createElement("optgroup"); g.label = cat;
    SALES.measureCats[cat].forEach(function(m){
      var o = document.createElement("option"); o.value = m; o.textContent = m; g.appendChild(o);

    sel.appendChild(g);

  sel.value = "Brukinsa patient recruited";

  });
}

/* ---------- charts ---------- */
var C_ACTUAL = "var(--series-1)";
var C_TARGET = "var(--series-2)";

function showTip(box, html, x, y){
  var t = box.querySelector(".tip");
  if(!t){ t = document.createElement("div"); t.className = "tip"; box.appendChild(t); }
  t.innerHTML = html; t.hidden = false;
  var bw = box.clientWidth, tw = t.offsetWidth;
  t.style.left = Math.max(4, Math.min(x - tw/2, bw - tw - 4)) + "px";
  t.style.top  = Math.max(0, y) + "px";
}
function hideTip(box){ var t = box.querySelector(".tip"); if(t) t.hidden = true; }

function lineChart(box, series, opts){
  var W = Math.max(box.clientWidth || 640, 340), H = 260;
  var P = {t:16, r:46, b:30, l:46};
  var iw = W - P.l - P.r, ih = H - P.t - P.b;
  var all = [];
  series.forEach(function(s){ s.values.forEach(function(v){ if(v !== null) all.push(v); }); });
  var max = all.length ? Math.max.apply(null, all) : 1;
  if(max <= 0) max = 1;
  var step = Math.pow(10, Math.floor(Math.log10(max))) / 2 || 1;
  var top = Math.ceil(max / step) * step;
  var ticks = 4;

  var x = function(i){ return P.l + (iw * i) / (MONTHS.length - 1); };
  var y = function(v){ return P.t + ih - (ih * v) / top; };

  var svg = ['<svg viewBox="0 0 ' + W + ' ' + H + '" width="' + W + '" height="' + H + '" role="img" aria-label="' + esc(opts.aria) + '">'];
  svg.push('<g class="axis">');
  for(var k = 0; k <= ticks; k++){
    var v = (top / ticks) * k, yy = y(v);
    svg.push('<line class="gridline" x1="' + P.l + '" x2="' + (W - P.r) + '" y1="' + yy + '" y2="' + yy + '"/>');
    svg.push('<text x="' + (P.l - 8) + '" y="' + (yy + 3.5) + '" text-anchor="end">' + money(v) + '</text>');
  }
  MONTHS.forEach(function(m, i){
    svg.push('<text x="' + x(i) + '" y="' + (H - 10) + '" text-anchor="middle">' + m + '</text>');
  });
  svg.push('<line class="axisline" x1="' + P.l + '" x2="' + (W - P.r) + '" y1="' + y(0) + '" y2="' + y(0) + '"/>');
  svg.push('</g>');

  series.forEach(function(s){
    var d = "", started = false, last = -1;
    s.values.forEach(function(v, i){
      if(v === null) return;
      d += (started ? "L" : "M") + x(i) + " " + y(v) + " ";
      started = true; last = i;
    });
    if(d) svg.push('<path d="' + d.trim() + '" fill="none" stroke="' + s.color + '" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"/>');
    s.values.forEach(function(v, i){
      if(v === null) return;
      svg.push('<circle cx="' + x(i) + '" cy="' + y(v) + '" r="4.5" fill="' + s.color + '" stroke="var(--paper)" stroke-width="2"/>');
    });
    if(last >= 0){
      svg.push('<text class="dlabel" x="' + (x(last) + 8) + '" y="' + (y(s.values[last]) + 4) + '" fill="' + s.color + '">' + money(s.values[last]) + '</text>');
    }
  });

  /* invisible hit columns for the crosshair */
  MONTHS.forEach(function(m, i){
    svg.push('<rect data-i="' + i + '" x="' + (x(i) - iw / (MONTHS.length * 2)) + '" y="' + P.t +
             '" width="' + (iw / MONTHS.length) + '" height="' + ih + '" fill="transparent"/>');
  });
  svg.push('</svg>');
  box.innerHTML = svg.join("");

  box.querySelectorAll("rect[data-i]").forEach(function(r){
    r.addEventListener("mouseenter", function(){
      var i = +r.dataset.i;
      var rows = series.map(function(s){
        return '<div class="row2"><span class="swatch" style="background:' + s.color + '"></span>' +
               esc(s.name) + ' <b>' + money(s.values[i]) + '</b></div>';
      }).join("");
      showTip(box, '<b>' + MONTHS[i] + '</b>' + rows, x(i), 8);
    });
    r.addEventListener("mouseleave", function(){ hideTip(box); });
  });
  box.addEventListener("mouseleave", function(){ hideTip(box); });
}

function barChart(box, rows, opts){
  var W = Math.max(box.clientWidth || 640, 320);
  var rowH = 26, P = {t:6, r:56, b:6, l:Math.min(190, Math.round(W * 0.34))};
  var H = P.t + P.b + rows.length * rowH;
  var iw = W - P.l - P.r;
  var max = Math.max.apply(null, rows.map(function(r){ return r.value || 0; }).concat([1]));
  var svg = ['<svg viewBox="0 0 ' + W + ' ' + H + '" width="' + W + '" height="' + H + '" role="img" aria-label="' + esc(opts.aria) + '">'];
  rows.forEach(function(r, i){
    var yy = P.t + i * rowH, bh = 14;
    var w = r.value ? Math.max(2, (iw * r.value) / max) : 0;
    svg.push('<text class="axis" x="' + (P.l - 10) + '" y="' + (yy + bh) + '" text-anchor="end" fill="var(--ink-2)" font-size="11.5">' + esc(r.label) + '</text>');
    svg.push('<rect data-i="' + i + '" x="' + P.l + '" y="' + (yy + 4) + '" width="' + w + '" height="' + bh + '" rx="4" fill="' + opts.color + '"/>');
    svg.push('<text class="dlabel" x="' + (P.l + w + 8) + '" y="' + (yy + bh - 1) + '" fill="var(--ink-2)">' + money(r.value) + '</text>');
  });
  svg.push('</svg>');
  box.innerHTML = svg.join("");
  box.querySelectorAll("rect[data-i]").forEach(function(rc){
    rc.addEventListener("mouseenter", function(){
      var r = rows[+rc.dataset.i];
      var bb = rc.getBoundingClientRect(), pb = box.getBoundingClientRect();
      showTip(box, '<b>' + esc(r.label) + '</b><br>' + esc(opts.measure) + ' <b>' + money(r.value) + '</b>' +
                   (r.sub ? '<br>' + esc(r.sub) : ''), bb.left - pb.left + bb.width, bb.top - pb.top - 6);
    });
    rc.addEventListener("mouseleave", function(){ hideTip(box); });
  });
}

/* ---------- sales page ---------- */
function renderSales(){
  var list = filteredAccounts();
  var measure = document.getElementById("fmeasure").value;
  var mv = document.getElementById("fmonth").value;
  var monthIdx = mv === "" ? "ytd" : +mv;
  var monthLabel = mv === "" ? "year to date" : MONTHS[monthIdx];
  var target = PAIRS[measure];

  document.getElementById("salessub").textContent =
    SALES.accounts.length + " accounts from " + SALES.source + ". " +
    list.length + " shown with the current filters.";

  /* tiles: the headline measures, not whatever is selected */
  var tiles = [
    {k:"Patients recruited", m:"Brukinsa patient recruited", t:"Brukinsa patient target"},
    {k:"New patients",       m:"Brukinsa New Pts recruited"},
    {k:"Sales (units)",      m:"Sales (units)", t:"Target (units)"},
    {k:"Consumption",        m:"Consumption"},
    {k:"Ending stock",       m:"Ending stock"},
    {k:"Accounts",           special:list.length}
  ];
  document.getElementById("tiles").innerHTML = tiles.map(function(t){
    if(t.special !== undefined){
      return '<div class="tile"><span class="k">' + esc(t.k) + '</span><span class="v">' + t.special +
             '</span><span class="s">of ' + SALES.accounts.length + ' in the file</span></div>';
    }
    var v = sumOf(list, t.m, monthIdx);
    var sub = "";
    if(t.t){
      var tv = sumOf(list, t.t, monthIdx);
      if(tv){
        var pct = Math.round((v / tv) * 100);
        sub = '<span class="s ' + (pct >= 100 ? "up" : "") + '">' + pct + '% of target ' + money(tv) + '</span>';
      }
    }
    return '<div class="tile"><span class="k">' + esc(t.k) + '</span><span class="v">' + money(v) + '</span>' +
           (sub || '<span class="s">' + esc(monthLabel) + '</span>') + '</div>';
  }).join("");

  /* trend */
  var series = [{name:measure, color:C_ACTUAL, values:seriesOf(list, measure)}];
  if(target) series.push({name:target, color:C_TARGET, values:seriesOf(list, target)});
  document.getElementById("trendtitle").textContent = measure + " by month";
  document.getElementById("trendsub").textContent =
    "Every month in the file, summed across the " + list.length + " account" + (list.length === 1 ? "" : "s") + " selected.";
  document.getElementById("trendlegend").innerHTML = series.length > 1 ? series.map(function(s){
    return '<span><span class="swatch" style="background:' + s.color + '"></span>' + esc(s.name) + '</span>';
  }).join("") : "";
  lineChart(document.getElementById("trendbox"), series, {aria:measure + " by month"});

  /* by account */
  var rows = list.map(function(a){
    var m = a.measures[measure];
    var v = m ? ((monthIdx === "ytd") ? m.ytd : m.m[monthIdx]) : null;
    return {label:a.account, value:(v === null || v === undefined || isNaN(v)) ? 0 : v, sub:a.city + " · " + a.ps};
  }).sort(function(x, y){ return y.value - x.value; });
  document.getElementById("acctitle").textContent = measure + " by account";
  document.getElementById("accsub").textContent = "Highest first, " + monthLabel + ".";
  barChart(document.getElementById("accbox"), rows, {color:C_ACTUAL, measure:measure, aria:measure + " by account"});

  /* table view */
  var th = '<thead><tr><th class="txt">Account</th><th class="txt">City</th><th class="txt">Owner</th>' +
           MONTHS.map(function(m){ return '<th>' + m + '</th>'; }).join("") + '<th>YTD</th></tr></thead>';
  var tb = '<tbody>' + list.map(function(a){
    var m = a.measures[measure];
    return '<tr><td class="acct">' + esc(a.account) + '</td><td class="txt dim">' + esc(a.city) +
           '</td><td class="txt dim">' + esc(a.ps) + '</td>' +
           MONTHS.map(function(_, i){ return '<td>' + (m ? money(m.m[i]) : "—") + '</td>'; }).join("") +
           '<td><b>' + (m ? money(m.ytd) : "—") + '</b></td></tr>';
  }).join("") + '</tbody>';
  document.getElementById("salestable").innerHTML = th + tb;
}

function loadEntries(){
  try{ entries = JSON.parse(localStorage.getItem(ENTRY_KEY) || "[]") || []; }
  catch(e){ entries = []; }
  if(!Array.isArray(entries)) entries = [];
}
function saveEntries(){
  try{ localStorage.setItem(ENTRY_KEY, JSON.stringify(entries)); return true; }
  catch(e){ return false; }
}
function buildEntryForm(){
  fillSelect(document.getElementById("eaccount"),
    SALES.accounts.map(function(a){ return {value:a.id, label:a.account + " — " + a.city}; }), "Select an account");
  var sel = document.getElementById("emeasure");
  sel.innerHTML = '<option value="">Select a measure</option>';
  Object.keys(SALES.measureCats).forEach(function(cat){
    var g = document.createElement("optgroup"); g.label = cat;
    SALES.measureCats[cat].forEach(function(m){
      var o = document.createElement("option"); o.value = m; o.textContent = m; g.appendChild(o);
    });
    sel.appendChild(g);
  });
  fillSelect(document.getElementById("emonth"),
    MONTHS.map(function(m, i){ return {value:String(i), label:m}; }), "Select a month");

  document.getElementById("entryform").addEventListener("submit", function(e){
    e.preventDefault();
    var acc = document.getElementById("eaccount").value;
    var mea = document.getElementById("emeasure").value;
    var mon = document.getElementById("emonth").value;
    var val = document.getElementById("evalue").value;
    var note = document.getElementById("enote").value.trim();
    var err = document.getElementById("entryerr");
    function bad(msg, id){ err.textContent = msg; err.hidden = false; var el = document.getElementById(id); if(el) el.focus(); }
    if(!acc) return bad("Choose an account.", "eaccount");
    if(!mea) return bad("Choose a measure.", "emeasure");
    if(mon === "") return bad("Choose a month.", "emonth");
    if(val === "" || isNaN(+val)) return bad("Enter a number.", "evalue");
    err.hidden = true;

    var a = SALES.accounts.filter(function(x){ return x.id === acc; })[0];
    entries.push({
      id:"e" + Date.now(), account:a.account, city:a.city, measure:mea,
      month:MONTHS[+mon], value:+val, note:note,
      by:(user && user.name) || "", role:(user && user.role) || "",
      at:new Date().toISOString()
    });
    var ok = saveEntries();
    document.getElementById("entrysaved").textContent = ok ? "" : "This browser is blocking storage";
    document.getElementById("evalue").value = "";
    document.getElementById("enote").value = "";
    renderEntries();
    toast(ok ? "Entry saved on this device" : "Could not save on this device");
  });

  document.getElementById("exportentries").addEventListener("click", exportEntries);
  document.getElementById("entrylist").addEventListener("click", function(e){
    var b = e.target.closest ? e.target.closest("[data-del2]") : null;
    if(!b) return;
    entries = entries.filter(function(x){ return x.id !== b.dataset.del2; });
    saveEntries(); renderEntries(); toast("Entry deleted");
  });
}
function renderEntries(){
  var host = document.getElementById("entrylist");
  document.getElementById("entrycount").textContent = entries.length ? "(" + entries.length + ")" : "";
  if(!entries.length){
    host.innerHTML = '<div class="emptybox">Nothing entered on this device yet.</div>';
    return;
  }
  host.innerHTML = entries.slice().reverse().map(function(en){
    return '<div class="entry2"><div><div><b>' + esc(en.account) + '</b> · ' + esc(en.measure) + ' · ' + esc(en.month) + '</div>' +
           (en.note ? '<div style="font-size:13px;color:var(--ink-2);margin-top:3px;">' + esc(en.note) + '</div>' : '') +
           '<div class="meta2">' + esc(stamp(en.at)) + (en.by ? '  ·  ' + esc(en.by) + (en.role ? ' (' + esc(en.role) + ')' : '') : '') + '</div></div>' +
           '<div style="text-align:right;"><div class="val">' + money(en.value) + '</div>' +
           '<button class="delnote" type="button" data-del2="' + esc(en.id) + '">Delete</button></div></div>';
  }).join("");
}
function exportEntries(){
  var L = [];
  L.push("BeOne Gulf - sales entries export");
  L.push("Internal draft. Not MRC approved.");
  L.push("Exported: " + new Date().toString());
  if(user) L.push("Exported by: " + user.name + " <" + user.email + ">" + (user.role ? " — " + user.role : ""));
  L.push("");
  L.push("Account,City,Measure,Month,Value,Note,Entered by,Role,Entered at");
  entries.forEach(function(e){
    function q(s){ return '"' + String(s === undefined || s === null ? "" : s).replace(/"/g, '""') + '"'; }
    L.push([q(e.account), q(e.city), q(e.measure), q(e.month), e.value, q(e.note), q(e.by), q(e.role), q(e.at)].join(","));
  });
  if(!entries.length) L.push("(no entries)");
  var text = L.join("\n");
  var name = "beone-sales-entries-" + new Date().toISOString().slice(0, 10) + ".csv";
  var started = false;
  try{ blobDownload(text, name); started = true; }catch(e){ started = false; }
  showNotesInline(text, started ? "Entries downloaded. If nothing arrived, copy them below." : "Copy your entries below.");
}

function buildAio(){
  fillSelect(document.getElementById("aiocat"), Object.keys(SALES.measureCats), "All categories");
  document.getElementById("aiocat").addEventListener("change", renderAio);
  document.getElementById("aiosearch").addEventListener("input", renderAio);
}
function renderAio(){
  var q = document.getElementById("aiosearch").value.trim().toLowerCase();
  var cat = document.getElementById("aiocat").value;
  var terms = q ? q.split(/\s+/) : [];
  var th = '<thead><tr><th class="txt">Account</th><th class="txt">City</th><th class="txt">Measure</th>' +
           MONTHS.map(function(m){ return '<th>' + m + '</th>'; }).join("") + '<th>Total</th><th>YTD</th></tr></thead>';
  var body = [], shown = 0;
  SALES.accounts.forEach(function(a){
    var ms = Object.keys(a.measures).filter(function(m){
      if(cat && a.measures[m].cat !== cat) return false;
      if(!terms.length) return true;
      var hay = (a.account + " " + a.city + " " + a.ps + " " + m + " " + a.measures[m].cat).toLowerCase();
      return terms.every(function(t){ return hay.indexOf(t) !== -1; });
    });
    if(!ms.length) return;
    body.push('<tr class="grouprow"><td class="txt" colspan="' + (MONTHS.length + 5) + '">' +
              esc(a.account) + ' · ' + esc(a.city) + ' · ' + esc(a.category) + ' · ' + esc(a.ps) + '</td></tr>');
    ms.forEach(function(m){
      var d = a.measures[m]; shown++;
      body.push('<tr><td class="acct">' + esc(a.account) + '</td><td class="txt dim">' + esc(a.city) +
        '</td><td class="txt">' + esc(m) + '</td>' +
        MONTHS.map(function(_, i){ return '<td>' + money(d.m[i]) + '</td>'; }).join("") +
        '<td>' + money(d.total) + '</td><td><b>' + money(d.ytd) + '</b></td></tr>');
    });
  });
  document.getElementById("aiotable").innerHTML = th + '<tbody>' + (body.join("") ||
    '<tr><td class="txt" colspan="' + (MONTHS.length + 5) + '">Nothing matches.</td></tr>') + '</tbody>';
  document.getElementById("aiosub").textContent =
    SALES.accounts.length + " accounts, " + Object.keys(SALES.measureCats).length +
    " measure categories, every month as recorded in " + SALES.source + ".";
  document.getElementById("aionote").textContent = shown + " rows shown. Figures are exactly as they appear in the source file.";
}

/* ---------- module switching ---------- */
function showModule(name){
  ["home","objections","sales","entry","allinone"].forEach(function(m){
    var el = document.getElementById("mod-" + m);
    if(el) el.hidden = (m !== name);
  });
  document.querySelectorAll("#modnav button").forEach(function(b){
    b.setAttribute("aria-current", b.dataset.mod === name ? "true" : "false");
  });
  var ex = document.getElementById("export");
  if(ex) ex.hidden = (name !== "objections");
  if(name === "home") renderHome();
  if(name === "sales") renderSales();
  if(name === "entry") renderEntries();
  if(name === "allinone") renderAio();
}

function startSales(){
  buildSalesFilters();
  buildEntryForm();
  buildAio();
  loadEntries();
  document.body.addEventListener("click", function(e){
    var b = e.target.closest ? e.target.closest("button[data-mod]") : null;
    if(b) showModule(b.dataset.mod);
  });
  document.getElementById("brandhome").addEventListener("click", function(){ showModule("home"); });
  var rt;
  window.addEventListener("resize", function(){
    clearTimeout(rt);
    rt = setTimeout(function(){
      if(!document.getElementById("mod-sales").hidden) renderSales();
    }, 180);
  });
}
