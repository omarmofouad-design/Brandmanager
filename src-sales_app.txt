/* An entry overrides the figure for that account, measure and month.
   YTD is carried through using the rule the source file itself uses for
   that account and measure: a Jan-Jul sum, the July value, or a figure
   the file states outright and that a monthly edit cannot re-derive. */
function overrideMap(){
  var o = {};
  entries.forEach(function(e){
    var mi = MONTHS.indexOf(e.month);
    if(mi < 0) return;
    o[e.account + "|" + e.measure + "|" + mi] = e.value;   /* last one wins */
  });
  return o;
}
var OVR = {};
function monthVal(a, measure, i){
  var k = a.account + "|" + measure + "|" + i;
  if(useEntries && OVR.hasOwnProperty(k)) return OVR[k];
  var m = a.measures[measure];
  return m ? m.m[i] : null;
}
function hasOverride(a, measure){
  if(!useEntries) return false;
  for(var i = 0; i < MONTHS.length; i++){
    if(OVR.hasOwnProperty(a.account + "|" + measure + "|" + i)) return true;
  }
  return false;
}
/* The workbook leaves the year-to-date cell empty for new patients, so
   the parser recorded no rule for it. Its own Total column is the sum of
   the months, so summing is the file's own arithmetic, not a new figure. */
var YTD_RULE = { "Brukinsa New Pts recruited": "sum" };

function ytdVal(a, measure){
  var m = a.measures[measure];
  if(!m) return null;
  var forced = YTD_RULE.hasOwnProperty(measure);
  if(!forced && !hasOverride(a, measure)) return m.ytd;
  if((forced ? YTD_RULE[measure] : m.k) === "sum"){
    var t = 0, seen = false;
    for(var i = 0; i <= SALES.ytdThrough; i++){
      var v = monthVal(a, measure, i);
      if(v !== null && v !== undefined && !isNaN(v)){ t += v; seen = true; }
    }
    return seen ? t : null;
  }
  if((forced ? YTD_RULE[measure] : m.k) === "last") return monthVal(a, measure, SALES.ytdThrough);
  return m.ytd;                      /* the file states it; leave it alone */
}
var useEntries = true;

function money(n){
  if(n === null || n === undefined || isNaN(n)) return "—";
  return (Math.round(n*100)/100).toLocaleString("en-GB");
}
function sumOf(list, measure, monthIdx){
  var t = 0, seen = false;
  list.forEach(function(a){
    if(!a.measures[measure]) return;
    var v = (monthIdx === "ytd") ? ytdVal(a, measure) : monthVal(a, measure, monthIdx);
    if(v !== null && v !== undefined && !isNaN(v)){ t += v; seen = true; }
  });
  return seen ? t : null;
}
function seriesOf(list, measure){
  return MONTHS.map(function(_, i){ return sumOf(list, measure, i); });
}
function measureList(){
  var out = [];
  Object.keys(SALES.measureCats).forEach(function(cat){
    SALES.measureCats[cat].forEach(function(m){ out.push({cat:cat, name:m}); });
  });
  return out;
}
function filteredAccounts(){
  var ctry = document.getElementById("fcountry").value;
  var own  = document.getElementById("fowner").value;
  var city = document.getElementById("fcity").value;
  return scopedAccounts().filter(function(a){
    if(ctry && a.country !== ctry) return false;
    if(own && a.ps !== own) return false;
    if(city && a.city !== city) return false;
    return true;
  });
}

/* ---------- filters ---------- */
function fillSelect(el, values, allLabel){
  el.innerHTML = "";
  if(allLabel){ var o = document.createElement("option"); o.value = ""; o.textContent = allLabel; el.appendChild(o); }
  values.forEach(function(v){
    var o = document.createElement("option");
    o.value = (typeof v === "string") ? v : v.value;
    o.textContent = (typeof v === "string") ? v : v.label;
    el.appendChild(o);
  });
}
function buildSalesFilters(){
  var ctries = Array.from(new Set(SALES.accounts.map(function(a){ return a.country; }))).sort();
  fillSelect(document.getElementById("fcountry"), ctries, "All countries");
  var owners = Array.from(new Set(SALES.accounts.map(function(a){ return a.ps; }))).sort();
  var cities = Array.from(new Set(SALES.accounts.map(function(a){ return a.city; }))).sort();
  fillSelect(document.getElementById("fowner"), owners, "All owners");
  fillSelect(document.getElementById("fcity"), cities, "All cities");

  var sel = document.getElementById("fmeasure");
  sel.innerHTML = "";
  Object.keys(SALES.measureCats).forEach(function(cat){
    var g = document.createElement("optgroup"); g.label = cat;
    SALES.measureCats[cat].forEach(function(m){
      var o = document.createElement("option"); o.value = m; o.textContent = m; g.appendChild(o);
    });
    sel.appendChild(g);
  });
  sel.value = "Brukinsa patient recruited";

  fillSelect(document.getElementById("fmonth"),
    MONTHS.map(function(m,i){ return {value:String(i), label:m}; }), "Year to date");

  buildAccountFilter();

  /* narrowing country, owner or city narrows the account list with it */
  ["fcountry","fowner","fcity"].forEach(function(id){
    document.getElementById(id).addEventListener("change", function(){
      buildAccountFilter();
      renderSales();
    });
  });
  ["faccount","fmeasure","fmonth"].forEach(function(id){
    document.getElementById(id).addEventListener("change", renderSales);
  });
}

/* The account list follows the other filters, and keeps the open account
   selected if it is still in range. */
function buildAccountFilter(){
  var el = document.getElementById("faccount");
  var was = el.value;
  var pool = filteredAccounts().slice().sort(function(a, b){
    return a.account.localeCompare(b.account);
  });
  fillSelect(el, pool.map(function(a){
    return { value:a.id, label:a.account + " — " + a.city };
  }), "All accounts (" + pool.length + ")");
  el.value = pool.some(function(a){ return a.id === was; }) ? was : "";
}
function openAccount(id){
  var el = document.getElementById("faccount");
  if(!Array.prototype.some.call(el.options, function(o){ return o.value === id; })){
    /* the account is outside the current filters — clear them and try again */
    document.getElementById("fcountry").value = "";
    document.getElementById("fowner").value = "";
    document.getElementById("fcity").value = "";
    buildAccountFilter();
  }
  el.value = id;
  renderSales();
  var m = document.getElementById("mod-sales");
  var p = m.querySelector(".scrollpane");
  if(p) p.scrollTop = 0;
}

function showTip(box, html, x, y){
  var t = box.querySelector(".tip");
  if(!t){ t = document.createElement("div"); t.className = "tip"; box.appendChild(t); }
  t.innerHTML = html; t.hidden = false;
  var bw = box.clientWidth, tw = t.offsetWidth;
  t.style.left = Math.max(4, Math.min(x - tw/2, bw - tw - 4)) + "px";
  t.style.top  = Math.max(0, y) + "px";
}
function hideTip(box){ var t = box.querySelector(".tip"); if(t) t.hidden = true; }

function lineChart(box, series, opts){
  var W = Math.max(box.clientWidth || 640, 340), H = 260;
  var P = {t:16, r:46, b:30, l:46};
  var iw = W - P.l - P.r, ih = H - P.t - P.b;
  var all = [];
  series.forEach(function(s){ s.values.forEach(function(v){ if(v !== null) all.push(v); }); });
  if(opts.ref && opts.ref.value > 0) all.push(opts.ref.value);
  var max = all.length ? Math.max.apply(null, all) : 1;
  var min = all.length ? Math.min.apply(null, all) : 0;
  if(max <= 0) max = 1;
  var step = Math.pow(10, Math.floor(Math.log10(Math.max(max, -min) || 1))) / 2 || 1;
  var top = Math.ceil(max / step) * step;
  /* stock corrections and the like go negative — the scale has to follow */
  var base = min < 0 ? -Math.ceil(-min / step) * step : 0;
  if(top <= base) top = base + step;
  var span = top - base;
  var ticks = 4;

  var x = function(i){ return P.l + (iw * i) / (MONTHS.length - 1); };
  var y = function(v){ return P.t + ih - (ih * (v - base)) / span; };

  var svg = ['<svg viewBox="0 0 ' + W + ' ' + H + '" width="' + W + '" height="' + H + '" role="img" aria-label="' + esc(opts.aria) + '">'];
  svg.push('<g class="axis">');
  for(var k = 0; k <= ticks; k++){
    var v = base + (span / ticks) * k, yy = y(v);
    svg.push('<line class="gridline" x1="' + P.l + '" x2="' + (W - P.r) + '" y1="' + yy + '" y2="' + yy + '"/>');
    svg.push('<text x="' + (P.l - 8) + '" y="' + (yy + 3.5) + '" text-anchor="end">' + money(v) + '</text>');
  }
  MONTHS.forEach(function(m, i){
    svg.push('<text x="' + x(i) + '" y="' + (H - 10) + '" text-anchor="middle">' + m + '</text>');
  });
  svg.push('<line class="axisline" x1="' + P.l + '" x2="' + (W - P.r) + '" y1="' + y(0) + '" y2="' + y(0) + '"/>');
  svg.push('</g>');
  if(opts.ref && opts.ref.value > 0 && opts.ref.value <= top){
    var ry = y(opts.ref.value);
    svg.push('<line class="refline" x1="' + P.l + '" x2="' + (W - P.r) + '" y1="' + ry + '" y2="' + ry + '"/>');
    svg.push('<text class="reflabel" x="' + (W - P.r) + '" y="' + (ry - 6) + '" text-anchor="end">' +
             esc(opts.ref.label) + '</text>');
  }

  series.forEach(function(s){
    /* a month with no figure breaks the line rather than being bridged */
    var d = "", pen = false, last = -1;
    s.values.forEach(function(v, i){
      if(v === null || v === undefined || isNaN(v)){ pen = false; return; }
      d += (pen ? "L" : "M") + x(i) + " " + y(v) + " ";
      pen = true; last = i;
    });
    if(d) svg.push('<path d="' + d.trim() + '" fill="none" stroke="' + s.color + '" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"/>');
    s.values.forEach(function(v, i){
      if(v === null) return;
      svg.push('<circle cx="' + x(i) + '" cy="' + y(v) + '" r="4.5" fill="' + s.color + '" stroke="var(--paper)" stroke-width="2"/>');
    });
    if(last >= 0){
      var ly = Math.max(P.t + 9, Math.min(y(s.values[last]) + 4, P.t + ih - 2));
      svg.push('<text class="dlabel" x="' + (x(last) + 8) + '" y="' + ly + '" fill="' + s.color + '">' + money(s.values[last]) + '</text>');
    }
  });

  /* invisible hit columns for the crosshair */
  MONTHS.forEach(function(m, i){
    svg.push('<rect data-i="' + i + '" x="' + (x(i) - iw / (MONTHS.length * 2)) + '" y="' + P.t +
             '" width="' + (iw / MONTHS.length) + '" height="' + ih + '" fill="transparent"/>');
  });
  svg.push('</svg>');
  box.innerHTML = svg.join("");

  box.querySelectorAll("rect[data-i]").forEach(function(r){
    r.addEventListener("mouseenter", function(){
      var i = +r.dataset.i;
      var rows = series.map(function(s){
        return '<div class="row2"><span class="swatch" style="background:' + s.color + '"></span>' +
               esc(s.name) + ' <b>' + money(s.values[i]) + '</b></div>';
      }).join("");
      showTip(box, '<b>' + MONTHS[i] + '</b>' + rows, x(i), 8);
    });
    r.addEventListener("mouseleave", function(){ hideTip(box); });
  });
  box.addEventListener("mouseleave", function(){ hideTip(box); });
}

function barChart(box, rows, opts){
  var W = Math.max(box.clientWidth || 640, 320);
  var rowH = 26, P = {t:6, r:56, b:6, l:Math.min(190, Math.round(W * 0.34))};
  var H = P.t + P.b + rows.length * rowH;
  var iw = W - P.l - P.r;
  var max = Math.max.apply(null, rows.map(function(r){ return r.value || 0; })
              .concat(rows.map(function(r){ return r.target || 0; })).concat([1]));
  /* the label column is fixed, so trim a long name to fit it — the full name stays in the tooltip */
  var maxCh = Math.max(8, Math.floor((P.l - 14) / 6.95));
  var fit = function(t){
    if(t.length <= maxCh) return t;
    var head = Math.ceil((maxCh - 1) * 0.55);
    return t.slice(0, head).replace(/[ ,\-]+$/, "") + "\u2026" + t.slice(-(maxCh - 1 - head));
  };
  var svg = ['<svg viewBox="0 0 ' + W + ' ' + H + '" width="' + W + '" height="' + H + '" role="img" aria-label="' + esc(opts.aria) + '">'];
  rows.forEach(function(r, i){
    var yy = P.t + i * rowH, bh = 14;
    var w = r.value > 0 ? Math.max(2, (iw * r.value) / max) : 0;   /* negatives read from the number */
    var lab = fit(r.label);
    svg.push('<text class="axis" x="' + (P.l - 10) + '" y="' + (yy + bh) + '" text-anchor="end" fill="var(--ink-2)" font-size="11.5"><title>' + esc(r.label) + '</title>' + esc(lab) + '</text>');
    svg.push('<rect data-i="' + i + '" x="' + P.l + '" y="' + (yy + 4) + '" width="' + w + '" height="' + bh + '" rx="4" fill="' + opts.color + '"/>');
    if(r.target !== null && r.target !== undefined && r.target > 0){
      var tx = P.l + (iw * r.target) / max;
      svg.push('<line class="tgtmark" x1="' + tx + '" x2="' + tx + '" y1="' + (yy + 1) +
               '" y2="' + (yy + bh + 7) + '"/>');
    }
    svg.push('<text class="dlabel" x="' + (P.l + Math.max(w, r.target ? (iw * r.target) / max : 0) + 8) +
             '" y="' + (yy + bh - 1) + '" fill="var(--ink-2)">' + money(r.value) + '</text>');
  });
  svg.push('</svg>');
  box.innerHTML = svg.join("");
  box.querySelectorAll("rect[data-i]").forEach(function(rc){
    rc.addEventListener("mouseenter", function(){
      var r = rows[+rc.dataset.i];
      var bb = rc.getBoundingClientRect(), pb = box.getBoundingClientRect();
      showTip(box, '<b>' + esc(r.label) + '</b><br>' + esc(opts.measure) + ' <b>' + money(r.value) + '</b>' +
                   (r.target ? '<br>Target <b>' + money(r.target) + '</b>' : '') +
                   (r.sub ? '<br>' + esc(r.sub) : ''), bb.left - pb.left + bb.width, bb.top - pb.top - 6);
    });
    rc.addEventListener("mouseleave", function(){ hideTip(box); });
  });
}

/* ---------- account field notes ---------- */
var ACCT_NOTES_KEY = "beone_account_notes_v1";
var anotes = {};
function loadAccountNotes(){
  try{ anotes = JSON.parse(localStorage.getItem(ACCT_NOTES_KEY) || "{}") || {}; }
  catch(e){ anotes = {}; }
  Object.keys(anotes).forEach(function(k){ if(!Array.isArray(anotes[k])) anotes[k] = []; });
}
function saveAccountNotes(){
  try{ localStorage.setItem(ACCT_NOTES_KEY, JSON.stringify(anotes)); return true; }
  catch(e){ return false; }
}
function acctNotesFor(id){ return Array.isArray(anotes[id]) ? anotes[id] : []; }
function acctNoteLogHTML(id){
  var list = acctNotesFor(id);
  if(!list.length) return '<li class="noneyet">No notes on this account yet.</li>';
  var h = "";
  for(var i = list.length - 1; i >= 0; i--){          /* newest first */
    h += '<li class="noteitem">'
       + '<div class="notehead"><time>' + esc(stamp(list[i].t)) + (list[i].who ? '  ·  ' + esc(list[i].who) : '') + '</time>'
       + '<button class="delnote" type="button" data-adel="' + i + '" aria-label="Delete this note">Delete</button></div>'
       + '<p>' + esc(list[i].text) + '</p></li>';
  }
  return h;
}
function addAccountNote(){
  var a = selectedAccount();
  if(!a) return;
  var box = document.getElementById("anoteinput");
  var text = box.value.trim();
  if(!text){ box.focus(); return; }
  if(!Array.isArray(anotes[a.id])) anotes[a.id] = [];
  anotes[a.id].push({ t:new Date().toISOString(), text:text,
                      who:(user && user.name) || "", role:(user && user.role) || "" });
  var ok = saveAccountNotes();
  box.value = "";
  document.getElementById("anotelog").innerHTML = acctNoteLogHTML(a.id);
  document.getElementById("asavedmsg").textContent = ok ? "" : "This browser is blocking storage";
  toast(ok ? "Note added to " + a.account : "Could not save on this device");

}
function deleteAccountNote(i){
  var a = selectedAccount();
  if(!a) return;
  var list = acctNotesFor(a.id);
  if(i < 0 || i >= list.length) return;
  list.splice(i, 1);
  saveAccountNotes();
  document.getElementById("anotelog").innerHTML = acctNoteLogHTML(a.id);
  toast("Note deleted");
}
/* appended to the export by notesText() */
function accountNotesText(){
  var L = [], n = 0, accs = 0;
  SALES.accounts.forEach(function(a){
    var list = acctNotesFor(a.id);
    if(!list.length) return;
    accs++;
    L.push("----------------------------------------------------------");
    L.push("ACCOUNT: " + a.account);
    L.push("         " + a.city + ", " + a.country + "  ·  owner " + a.ps + "  ·  " + a.category);
    L.push("");
    list.forEach(function(en){
      n++;
      L.push("  " + stamp(en.t) + (en.who ? "  ·  " + en.who + (en.role ? " (" + en.role + ")" : "") : ""));
      en.text.split("\n").forEach(function(line){ L.push("    " + line); });
      L.push("");
    });
  });
  if(!n) return "";
  return "\n\n==========================================================\n" +
         "ACCOUNT FIELD NOTES\n" +
         "==========================================================\n" +
         L.join("\n") +
         "----------------------------------------------------------\n" +
         "Total: " + n + " note" + (n === 1 ? "" : "s") + " across " + accs +
         " account" + (accs === 1 ? "" : "s");
}

/* ---------- derived figures ---------- */
/* Therapy columns exactly as the file names them. Nothing is renamed,
   combined or recalculated; each bar is that column's own figure. */
var THERAPIES = ["Brukinsa patient recruited", "Acala CLL", "Acala MCL", "A+V",
                 "Ibrutinib mono", "I+V", "Pirtubrutinb", "Ibrutinib"];

function therapyRows(list, monthIdx){
  return THERAPIES.map(function(m){
    var v = sumOf(list, m, monthIdx);
    return { label:(m === "Brukinsa patient recruited" ? "BRUKINSA" : m),
             value:(v === null || isNaN(v)) ? 0 : v,
             sub:(m === "Brukinsa patient recruited" ? "Brukinsa patient recruited" : m) };
  }).sort(function(a, b){ return b.value - a.value; });
}
/* Brukinsa as a share of every therapy column added together. A ratio of
   the file's own figures — the file does not state this number itself. */
function brukShare(list, monthIdx){
  var tot = 0, bru = sumOf(list, "Brukinsa patient recruited", monthIdx);
  THERAPIES.forEach(function(m){
    var v = sumOf(list, m, monthIdx);
    if(v !== null && !isNaN(v)) tot += v;
  });
  if(!tot || bru === null) return null;
  return (bru / tot) * 100;
}
/* Ending stock measured against the average month of consumption. */
function stockCover(list){
  var stock = sumOf(list, "Ending stock", SALES.ytdThrough);
  var t = 0, n = 0;
  for(var i = 0; i <= SALES.ytdThrough; i++){
    var v = sumOf(list, "Consumption", i);
    if(v !== null && !isNaN(v)){ t += v; n++; }
  }
  /* a stock recorded below zero cannot be expressed as months of cover */
  if(!n || !t || stock === null || stock < 0) return null;
  return stock / (t / n);
}

/* ---------- attainment chart ---------- */
/* Percent of target per account, against a 100% reference line. The
   number is written on every bar, so the bands are never colour alone. */
function attainChart(box, rows, opts){
  var W = Math.max(box.clientWidth || 640, 320);
  var rowH = 26, P = {t:20, r:58, b:6, l:Math.min(190, Math.round(W * 0.34))};
  var H = P.t + P.b + rows.length * rowH;
  var iw = W - P.l - P.r;
  /* the scale stops at 150% so one runaway account cannot squash the rest;
     a bar past it runs full width and keeps its true number as its label */
  var top = Math.min(150, Math.max(120, Math.ceil(Math.max.apply(null,
              rows.map(function(r){ return r.pct; }).concat([100])) / 20) * 20));
  var x = function(v){ return P.l + (iw * Math.min(v, top)) / top; };
  var maxCh = Math.max(8, Math.floor((P.l - 14) / 6.95));
  var fit = function(t){
    if(t.length <= maxCh) return t;
    var head = Math.ceil((maxCh - 1) * 0.55);
    return t.slice(0, head).replace(/[ ,\-]+$/, "") + "…" + t.slice(-(maxCh - 1 - head));
  };
  var band = function(p){ return p >= 100 ? "var(--ok)" : (p >= 70 ? "var(--warn)" : "var(--stop)"); };

  var s = ['<svg viewBox="0 0 ' + W + ' ' + H + '" width="' + W + '" height="' + H +
           '" role="img" aria-label="' + esc(opts.aria) + '">'];
  s.push('<line class="refline" x1="' + x(100) + '" x2="' + x(100) + '" y1="' + (P.t - 6) +
         '" y2="' + (H - P.b) + '"/>');
  s.push('<text class="reflabel" x="' + x(100) + '" y="' + (P.t - 10) + '" text-anchor="middle">target</text>');
  rows.forEach(function(r, i){
    var yy = P.t + i * rowH, bh = 14;
    var w = Math.max(2, x(r.pct) - P.l);
    s.push('<text class="axis" x="' + (P.l - 10) + '" y="' + (yy + bh) + '" text-anchor="end" ' +
           'fill="var(--ink-2)" font-size="11.5"><title>' + esc(r.label) + '</title>' + esc(fit(r.label)) + '</text>');
    s.push('<rect data-i="' + i + '" x="' + P.l + '" y="' + (yy + 4) + '" width="' + w +
           '" height="' + bh + '" rx="4" fill="' + band(r.pct) + '"/>');
    s.push('<text class="dlabel" x="' + (P.l + w + 8) + '" y="' + (yy + bh - 1) +
           '" fill="var(--ink-2)">' + Math.round(r.pct) + '%</text>');
  });
  s.push('</svg>');
  box.innerHTML = s.join("");
  box.querySelectorAll("rect[data-i]").forEach(function(rc){
    rc.addEventListener("mouseenter", function(){
      var r = rows[+rc.dataset.i];
      var bb = rc.getBoundingClientRect(), pb = box.getBoundingClientRect();
      showTip(box, '<b>' + esc(r.label) + '</b><br>' + esc(opts.measure) + ' <b>' + money(r.actual) +

              bb.left - pb.left + bb.width, bb.top - pb.top - 6);
    });

  });
}

function renderAllAccounts(list, universe, measure, monthIdx, monthLabel, target){
  var cover = stockCover(list);
  var endStock = sumOf(list, "Ending stock", SALES.ytdThrough);
  var share = brukShare(list, monthIdx);
  var pat = sumOf(list, "Brukinsa patient recruited", monthIdx);
  var patT = sumOf(list, "Brukinsa patient target", monthIdx);
  var un = sumOf(list, "Sales (units)", monthIdx);
  var unT = sumOf(list, "Target (units)", monthIdx);
  var newPts = sumOf(list, "Brukinsa New Pts recruited", monthIdx);
  var pct = function(v, t){ return (t && v !== null) ? Math.round((v / t) * 100) : null; };
  var pp = pct(pat, patT), up = pct(un, unT);

  document.getElementById("tiles").innerHTML =
    tileHTML("Accounts", list.length, "of " + universe.length + " in view") +
    tileHTML("Patients recruited", money(pat),
      patT ? (pp + "% of target " + money(patT)) : esc(monthLabel), pp >= 100) +

      unT ? (up + "% of target " + money(unT)) : esc(monthLabel), up >= 100) +
    tileHTML("New patients", money(newPts), esc(monthLabel)) +
    tileHTML("Ending stock", money(endStock),
      (endStock !== null && endStock < 0) ? "recorded below zero in the file"
        : cover === null ? "as at " + MONTHS[SALES.ytdThrough]
                         : (Math.round(cover * 10) / 10) + " months of cover") +
    tileHTML("Share of therapies", share === null ? "&mdash;" : Math.round(share) + "%",
      "of every therapy column");

  /* trend */
  var series = [{name:measure, color:C_ACTUAL, values:seriesOf(list, measure)}];

  lineChart(document.getElementById("trendbox"), series, {aria:measure + " by month"});

  var att = document.getElementById("attcard");
  if(target){
    var arows = list.map(function(a){
      var v = (monthIdx === "ytd") ? ytdVal(a, measure) : monthVal(a, measure, monthIdx);
      var t = (monthIdx === "ytd") ? ytdVal(a, target)  : monthVal(a, target, monthIdx);
      if(!t || t <= 0 || v === null || v === undefined || isNaN(v)) return null;
      return { label:a.account, actual:v, target:t, pct:(v / t) * 100 };
    }).filter(Boolean).sort(function(x, y){ return x.pct - y.pct; });

    if(arows.length){
      att.hidden = false;
      var behind = arows.filter(function(r){ return r.pct < 70; }).length;
      var shown = arows.slice(0, 20);
      document.getElementById("atttitle").textContent = measure + " against target, by account";
      setSub("attsub", "Worst first, " + monthLabel + ". " +
        (behind ? behind + " of " + arows.length + " below 70%." : "None below 70%.") +

      document.getElementById("attlegend").innerHTML =

        '<span><span class="swatch" style="background:var(--warn)"></span>70&ndash;99%</span>' +

               value:(v === null || isNaN(v)) ? 0 : v,
               target:planTarget(a),
               sub:a.country + " · " + a.ps };
    }).sort(function(x, y){ return (x.value - x.target) - (y.value - y.target); });  /* furthest short, first */
    var got = nrows.reduce(function(t, r){ return t + r.value; }, 0);
    var met = nrows.filter(function(r){ return r.value >= r.target; }).length;
    setSub("newsub", "Furthest short first. " + got + " of " + plan.target + " planned for the year, " +
      met + " of " + plan.accounts + " account" + (plan.accounts === 1 ? "" : "s") + " already there.");

      '<span><span class="swatch" style="background:' + C_ACTUAL + '"></span>New patients so far</span>' + TICK_LEGEND;
    barChart(document.getElementById("newbox"), nrows,
      {color:C_ACTUAL, measure:"New patients", aria:"New patients against the 2026 plan, by account"});
    document.getElementById("newnote").textContent = "Plan from " + PLAN2026.source + ".";
  } else { ncard.hidden = true; }

  /* by account */
  var rows = list.map(function(a){
    var v = a.measures[measure] ? ((monthIdx === "ytd") ? ytdVal(a, measure) : monthVal(a, measure, monthIdx)) : null;
    var t = (target && a.measures[target])
      ? ((monthIdx === "ytd") ? ytdVal(a, target) : monthVal(a, target, monthIdx)) : null;
    return {label:a.account + (hasOverride(a, measure) ? " *" : ""),
            value:(v === null || v === undefined || isNaN(v)) ? 0 : v,
            target:(t === null || t === undefined || isNaN(t)) ? null : t,
            sub:a.country + " · " + a.city + " · " + a.ps};
  }).sort(function(x, y){ return y.value - x.value; });
  document.getElementById("acctitle").textContent = measure + " by account";
  setSub("accsub", "Highest first, " + monthLabel + ".");
  if(rows.length) barChart(document.getElementById("accbox"), rows, {color:C_ACTUAL, measure:measure, aria:measure + " by account"});
  else document.getElementById("accbox").innerHTML =
    '<div class="emptybox">No accounts match these filters. Clear them, or switch to <b>Everyone</b> above.</div>';

  /* therapies */
  var trows = therapyRows(list, monthIdx).filter(function(r){ return r.value > 0; });
  setSub("thersub", monthLabel.charAt(0).toUpperCase() + monthLabel.slice(1) + ", each column as recorded.");
  if(trows.length) barChart(document.getElementById("therbox"), trows,
    {color:C_ACTUAL, measure:"Patients", aria:"Patients by therapy"});
  else document.getElementById("therbox").innerHTML =
    '<div class="emptybox">No therapy figures recorded for the accounts in view.</div>';

  /* table */
  var th = '<thead><tr><th class="txt">Account</th><th class="txt">Country</th><th class="txt">City</th><th class="txt">Owner</th>' +
           MONTHS.map(function(m){ return '<th>' + m + '</th>'; }).join("") + '<th>YTD</th></tr></thead>';
  var tb = '<tbody>' + list.map(function(a){
    var has = a.measures[measure];
    return '<tr class="acctrow" data-acc="' + esc(a.id) + '"><td class="acct">' + esc(a.account) + (hasOverride(a, measure) ? ' *' : '') +
           '</td><td class="txt dim">' + esc(a.country) + '</td><td class="txt dim">' + esc(a.city) +
           '</td><td class="txt dim">' + esc(a.ps) + '</td>' +
           MONTHS.map(function(_, i){ return '<td>' + (has ? money(monthVal(a, measure, i)) : "—") + '</td>'; }).join("") +
           '<td><b>' + (has ? money(ytdVal(a, measure)) : "—") + '</b></td></tr>';
  }).join("") + '</tbody>';
  document.getElementById("salestable").innerHTML = th + tb;
  document.getElementById("salesnote").textContent = "Source: " + SALES.source + ".";
}

function exportEntries(){
  var L = [];
  L.push("BeOne Gulf - sales entries export");
  L.push("Internal draft. Not MRC approved.");
  L.push("Exported: " + new Date().toString());
  if(user) L.push("Exported by: " + user.name + " <" + user.email + ">" + (user.role ? " — " + user.role : ""));
  L.push("");
  L.push("Account,City,Measure,Month,Value,Note,Entered by,Role,Entered at");
  entries.forEach(function(e){
    function q(s){ return '"' + String(s === undefined || s === null ? "" : s).replace(/"/g, '""') + '"'; }
    L.push([q(e.account), q(e.city), q(e.measure), q(e.month), e.value, q(e.note), q(e.by), q(e.role), q(e.at)].join(","));
  });
  if(!entries.length) L.push("(no entries)");
  var text = L.join("\n");
  var name = "beone-sales-entries-" + new Date().toISOString().slice(0, 10) + ".csv";
  var started = false;
  try{ blobDownload(text, name); started = true; }catch(e){ started = false; }
  showNotesInline(text, started ? "Entries downloaded. If nothing arrived, copy them below." : "Copy your entries below.");
}

/* ---------- module switching ----------
   Home is the launcher. Opening a section adds a tab; the bar only
   ever shows sections that are actually open. Back walks the trail
   of views, Home returns to the launcher without closing anything. */
var MOD_LABEL = {
  objections:"Objection handler", sales:"Sales",
  entry:"Enter sales", activities:"Activities"
};
var MOD_IDS = ["home","objections","sales","entry","activities"];
var openTabs = [];
var navStack = [];
var current  = "home";

var SVG_BACK = '<svg viewBox="0 0 24 24" width="16" height="16" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round"><path d="M15 5l-7 7 7 7"/></svg>';
var SVG_HOME = '<svg viewBox="0 0 24 24" width="16" height="16" fill="none" stroke="currentColor" stroke-width="2.1" stroke-linecap="round" stroke-linejoin="round"><path d="M3 11l9-7 9 7"/><path d="M5.5 9.5V20h13V9.5"/></svg>';

function renderNav(){
  var nav = document.getElementById("modnav");
  if(current === "home" && !openTabs.length){ nav.hidden = true; nav.innerHTML = ""; return; }
  nav.hidden = false;
  var h = '';
  h += '<button class="navbtn" type="button" data-nav="back"' + (navStack.length ? '' : ' disabled') +
       ' aria-label="Back">' + SVG_BACK + 'Back</button>';
  h += '<button class="navbtn" type="button" data-nav="home" aria-label="Home">' + SVG_HOME + 'Home</button>';
  if(openTabs.length) h += '<span class="navsep" aria-hidden="true"></span>';
  h += openTabs.map(function(m){
    return '<span class="tab"' + (current === m ? ' aria-current="true"' : '') + '>' +
      '<button class="tname" type="button" data-tab="' + m + '">' + esc(MOD_LABEL[m]) + '</button>' +
      '<button class="tclose" type="button" data-close="' + m + '" aria-label="Close ' + esc(MOD_LABEL[m]) + '">&times;</button>' +
      '</span>';
  }).join("");
  nav.innerHTML = h;
}

function showModule(name, fromBack){
  if(MOD_IDS.indexOf(name) === -1) return;
  if(!fromBack && name !== current) navStack.push(current);
  current = name;
  if(name !== "home" && openTabs.indexOf(name) === -1) openTabs.push(name);

  MOD_IDS.forEach(function(m){
    var el = document.getElementById("mod-" + m);
    if(el) el.hidden = (m !== name);
  });
  var ex = document.getElementById("export");
  if(ex) ex.hidden = (name !== "objections");

  if(name === "home")     renderHome();
  if(name === "sales")    renderSales();
  if(name === "entry")    renderEntries();
  if(name === "activities") renderActs();
  renderNav();
}

function goBack(){
  if(!navStack.length) return;
  var prev = navStack.pop();
  if(prev !== "home" && openTabs.indexOf(prev) === -1) prev = "home";
  showModule(prev, true);
}

function closeTab(name){
  openTabs = openTabs.filter(function(m){ return m !== name; });
  navStack = navStack.filter(function(m){ return m === "home" || openTabs.indexOf(m) !== -1; });
  if(current === name) showModule(openTabs.length ? openTabs[openTabs.length - 1] : "home", true);
  else renderNav();
}

function startSales(){
  loadEntries();
  loadAccountNotes();
  buildSalesFilters();
  buildEntryForm();
  buildActs();
  loadScope();
  renderScope();

  /* account detail: open, close, note keeping */
  document.getElementById("accall").addEventListener("click", function(){
    document.getElementById("faccount").value = "";
    renderSales();
  });
  document.getElementById("accentry").addEventListener("click", function(){
    var a = selectedAccount();
    showModule("entry");
    if(a){
      document.getElementById("eaccount").value = a.id;
      document.getElementById("emeasure").focus();
    }
  });
  document.getElementById("addanote").addEventListener("click", addAccountNote);
  document.getElementById("anoteinput").addEventListener("keydown", function(e){
    if(e.key === "Enter" && (e.metaKey || e.ctrlKey)){ e.preventDefault(); addAccountNote(); }
  });
  document.getElementById("anotelog").addEventListener("click", function(e){
    var d = e.target.closest ? e.target.closest("[data-adel]") : null;
    if(d) deleteAccountNote(parseInt(d.dataset.adel, 10));
  });
  document.getElementById("salestable").addEventListener("click", function(e){
    var r = e.target.closest ? e.target.closest("tr[data-acc]") : null;
    if(r) openAccount(r.dataset.acc);
  });

  document.body.addEventListener("click", function(e){
    var t = e.target;
    var close = t.closest ? t.closest("[data-close]") : null;
    if(close){ closeTab(close.dataset.close); return; }
    var tab = t.closest ? t.closest("[data-tab]") : null;
    if(tab){ showModule(tab.dataset.tab); return; }
    var sc = t.closest ? t.closest("[data-scope]") : null;
    if(sc){ setScope(sc.dataset.scope); return; }
    var nb = t.closest ? t.closest("[data-nav]") : null;
    if(nb){ nb.dataset.nav === "back" ? goBack() : showModule("home"); return; }
    var card = t.closest ? t.closest("button[data-mod]") : null;
    if(card) showModule(card.dataset.mod);
  });
  document.getElementById("brandhome").addEventListener("click", function(){ showModule("home"); });
  var rt;
  window.addEventListener("resize", function(){
    clearTimeout(rt);
    rt = setTimeout(function(){
      if(!document.getElementById("mod-sales").hidden) renderSales();
      if(!document.getElementById("mod-activities").hidden) renderActs();
    }, 180);
  });
}
